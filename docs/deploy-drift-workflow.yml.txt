name: Deploy drift

# "Merged" and "deployed" are not the same word, and on 17 August they came
# apart for most of a day with no signal: first Vercel's daily build allowance
# ran out, then production deployments stopped firing while PR previews kept
# succeeding — reporting nothing at all. The accessibility fixes existed and no
# user had them.
#
# This runs after every push to main (once the deployment has had time to land)
# and on a schedule, so the gap announces itself instead of being noticed by
# accident.

on:
  push:
    branches: [main]
  schedule:
    - cron: "23 */6 * * *"
  workflow_dispatch:

permissions:
  contents: read
  deployments: read

jobs:
  drift:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v5
        with:
          # The check compares main against the deployed sha, so it needs real
          # history — a shallow clone cannot answer "how far behind".
          fetch-depth: 0
      - uses: actions/setup-node@v5
        with:
          node-version: 20
      # A deployment takes a couple of minutes to report. Checking immediately
      # would flag every push as drift.
      - name: wait for the deployment to report
        if: github.event_name == 'push'
        run: sleep 240
      - run: npm run deploy:check
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
